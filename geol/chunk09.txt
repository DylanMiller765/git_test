const POMO_KEY='geol1304_pomodoro_v1';
const FOCUS_SECS=25*60, BREAK_SECS=5*60;
let P=JSON.parse(localStorage.getItem(POMO_KEY)||'null')||{subject:'GEOL',phase:'focus',left:FOCUS_SECS,running:false,endsAt:null};
let pomoTimer=null, pomoAudio=null;
function savePomo(){localStorage.setItem(POMO_KEY,JSON.stringify(P))}
function pomoTotal(){return P.phase==='focus'?FOCUS_SECS:BREAK_SECS}
function nextSubject(s){return s==='GEOL'?'Linear Algebra':'GEOL'}
function fmtTime(sec){sec=Math.max(0,Math.ceil(sec));return String(Math.floor(sec/60)).padStart(2,'0')+':'+String(sec%60).padStart(2,'0')}
function ensureAudio(){try{if(!pomoAudio)pomoAudio=new (window.AudioContext||window.webkitAudioContext)();if(pomoAudio.state==='suspended')pomoAudio.resume()}catch(e){}}
function beep(){try{if(!pomoAudio)return;let o=pomoAudio.createOscillator(),g=pomoAudio.createGain();o.connect(g);g.connect(pomoAudio.destination);o.frequency.value=740;g.gain.setValueAtTime(.12,pomoAudio.currentTime);g.gain.exponentialRampToValueAtTime(.001,pomoAudio.currentTime+.35);o.start();o.stop(pomoAudio.currentTime+.35)}catch(e){}}
function renderPomo(){
  if(!$('#pomoTime'))return;
  $('#pomoTime').textContent=fmtTime(P.left);
  $('#pomoPhase').textContent=P.phase==='focus'?'Focus':'Break';
  $('#pomoSubject').textContent=P.phase==='focus'?`${P.subject} focus`:`Break · next: ${nextSubject(P.subject)}`;
  $('#pomoStart').textContent=P.running?'Pause':'Start';
  $('#pomoNext').textContent=P.phase==='focus'?`Switch to ${nextSubject(P.subject)} →`:`Skip break → ${nextSubject(P.subject)}`;
  $('#pomoBar').style.width=Math.max(0,Math.min(100,(P.left/pomoTotal())*100))+'%';
  document.title=(P.running?fmtTime(P.left)+' · ':'')+'GEOL 1304 Test 1';
}
function finishPomo(){
  beep();
  if(P.phase==='focus'){
    P.phase='break';P.left=BREAK_SECS;P.running=true;P.endsAt=Date.now()+BREAK_SECS*1000;
  }else{
    P.subject=nextSubject(P.subject);P.phase='focus';P.left=FOCUS_SECS;P.running=false;P.endsAt=null;
  }
  savePomo();renderPomo();
}
function tickPomo(){
  if(P.running&&P.endsAt){P.left=Math.max(0,Math.ceil((P.endsAt-Date.now())/1000));if(P.left<=0){finishPomo();return}}
  renderPomo();
}
function startPomo(){
  ensureAudio();
  if(P.running){P.left=Math.max(0,Math.ceil((P.endsAt-Date.now())/1000));P.running=false;P.endsAt=null}
  else{P.running=true;P.endsAt=Date.now()+P.left*1000}
  savePomo();tickPomo();
}
function resetPomo(){P.phase='focus';P.left=FOCUS_SECS;P.running=false;P.endsAt=null;savePomo();renderPomo()}
function advancePomo(){
  if(P.phase==='break')P.subject=nextSubject(P.subject);else P.subject=nextSubject(P.subject);
  P.phase='focus';P.left=FOCUS_SECS;P.running=false;P.endsAt=null;savePomo();renderPomo();
}
if(P.running&&P.endsAt&&P.endsAt<=Date.now()){
  // Recover sensibly after a refresh/tab sleep.
  if(P.phase==='focus'){P.phase='break';P.left=BREAK_SECS;P.running=false;P.endsAt=null}
  else{P.subject=nextSubject(P.subject);P.phase='focus';P.left=FOCUS_SECS;P.running=false;P.endsAt=null}
  savePomo();
}
